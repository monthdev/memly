file(GLOB MEMLY_QML_ROOT CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.qml")
file(GLOB MEMLY_QML_COMPONENT CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Component/*.qml")
file(GLOB MEMLY_QML_SCREEN_CARD CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Screen/Card/*.qml")
file(GLOB MEMLY_QML_SECTION_ADD_CARD CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Section/AddCard/*.qml")
file(GLOB MEMLY_QML_SECTION_BROWSE CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Section/Browse/*.qml")
file(GLOB MEMLY_QML_SECTION_DECKS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Section/Decks/*.qml")
set(MEMLY_QML_FILES ${MEMLY_QML_ROOT} ${MEMLY_QML_COMPONENT} ${MEMLY_QML_SCREEN_CARD} ${MEMLY_QML_SECTION_ADD_CARD}
                    ${MEMLY_QML_SECTION_BROWSE} ${MEMLY_QML_SECTION_DECKS}
)
foreach(MEMLY_QML_FILE IN LISTS MEMLY_QML_FILES)
  file(RELATIVE_PATH MEMLY_QML_RESOURCE_ALIAS "${CMAKE_CURRENT_SOURCE_DIR}" "${MEMLY_QML_FILE}")
  set_source_files_properties("${MEMLY_QML_FILE}" PROPERTIES QT_RESOURCE_ALIAS "${MEMLY_QML_RESOURCE_ALIAS}")
endforeach()

qt_add_qml_module(
  memly_qml
  STATIC
  URI
  Memly
  VERSION
  1.0
  PLUGIN_TARGET
  memly_qml_plugin
  OUTPUT_DIRECTORY
  "${CMAKE_BINARY_DIR}/Memly"
  QML_FILES
  ${MEMLY_QML_FILES}
)

target_link_libraries(
  memly_qml
  PRIVATE Qt6::Quick Memly.Bridge memly_sanitizers
)
