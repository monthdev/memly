     "${CMAKE_CURRENT_SOURCE_DIR}/source/_Sql/Deck/Delete/*.sql"

     "${CMAKE_CURRENT_SOURCE_DIR}/source/_Sql/Deck/Insert/*.sql"

  ${MEMLY_REPOSITORY_PRIVATE_HEADERS}
  ${MEMLY_REPOSITORY_SQL_ACCESSOR_HEADERS}
  SOURCES
