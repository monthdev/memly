file(GLOB MEMLY_DATABASE_EXPORTED_MODULES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/export/*.cppm")
file(GLOB MEMLY_DATABASE_IMPLEMENTATION_UNITS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source/*.cpp")
file(GLOB MEMLY_DATABASE_INTERNAL_MODULES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source/Internal/*.cppm")
file(GLOB MEMLY_DATABASE_INTERNAL_IMPLEMENTATION_UNITS CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_SOURCE_DIR}/source/Internal/*.cpp"
)
file(GLOB MEMLY_DATABASE_BOOTSTRAP_SQL CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source/Sql/Bootstrap/*.sql")
file(GLOB MEMLY_DATABASE_INSERT_SQL CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source/Sql/Insert/*.sql")
file(GLOB MEMLY_DATABASE_SELECT_SQL CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source/Sql/Select/*.sql")
file(GLOB MEMLY_DATABASE_VERSION_SQL CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source/Sql/Version/*.sql")

  Memly.Database
  Memly.Database
  PUBLIC_MODULES
  ${MEMLY_DATABASE_EXPORTED_MODULES}
  PRIVATE_MODULES
  ${MEMLY_DATABASE_INTERNAL_MODULES}
  SOURCES
  ${MEMLY_DATABASE_IMPLEMENTATION_UNITS}
  ${MEMLY_DATABASE_INTERNAL_IMPLEMENTATION_UNITS}
)
memly_configure_duckdb_header(Memly.Database)

qt_add_resources(
  Memly.Database
  "DatabaseMigrationSql"
  PREFIX
  "/Sql"
  BASE
  "${CMAKE_CURRENT_SOURCE_DIR}/source/Sql"
  FILES
  ${MEMLY_DATABASE_BOOTSTRAP_SQL}
  ${MEMLY_DATABASE_INSERT_SQL}
  ${MEMLY_DATABASE_SELECT_SQL}
  ${MEMLY_DATABASE_VERSION_SQL}
)
